// ==UserScript== // @name MTR LOHAS Park Live Departures for GitHub Dashboard // @namespace https://github.com/ // @version 1.0 // @description Display real-time LOHAS Park MTR departure times on your GitHub dashboard // @author You // @match https://github.com/dashboard* // @match https://github.com/ // @grant GM_xmlhttpRequest // @connect rt.data.gov.hk // ==/UserScript== (function() { 'use strict'; const DEST_NAMES = { 'NOP': { en: 'North Point', zh: '北角' }, 'TIK': { en: 'Tiu Keng Leng', zh: '調景嶺' }, 'POA': { en: 'Po Lam', zh: '寶琳' } }; function createMTRWidget() { if (document.getElementById('mtr-lohas-widget')) return; // Container styled to match GitHub's dark/light UI const widget = document.createElement('div'); widget.id = 'mtr-lohas-widget'; widget.style.cssText = ` margin-bottom: 20px; padding: 16px; background: var(--bgColor-muted, #161b22); border: 1px solid var(--borderColor-default, #30363d); border-radius: 8px; color: var(--fgColor-default, #c9d1d9); font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Helvetica, Arial, sans-serif; font-size: 13px; `; widget.innerHTML = `
LOHAS Park (康城) Tseung Kwan O Line
Loading live departures...
`; // Insert into GitHub dashboard sidebar or feed const targetContainer = document.querySelector('aside') || document.querySelector('.news') || document.querySelector('#dashboard'); if (targetContainer) { targetContainer.insertBefore(widget, targetContainer.firstChild); } document.getElementById('mtr-refresh-btn').addEventListener('click', fetchSchedule); } function fetchSchedule() { const contentDiv = document.getElementById('mtr-schedule-content'); if (!contentDiv) return; GM_xmlhttpRequest({ method: "GET", url: "https://rt.data.gov.hk/v1/transport/mtr/getSchedule.php?line=TKL&sta=LHP", onload: function(response) { try { const data = JSON.parse(response.responseText); if (data.status === 0 || !data.data || !data.data['TKL-LHP']) { contentDiv.innerHTML = 'Service not available or outside operating hours.'; return; } const departures = data.data['TKL-LHP'].UP || []; if (departures.length === 0) { contentDiv.innerHTML = 'No upcoming departures.'; return; } let html = '
'; departures.forEach(train => { const dest = DEST_NAMES[train.dest] ? `${DEST_NAMES[train.dest].en} (${DEST_NAMES[train.dest].zh})` : train.dest; const ttnt = train.ttnt === '0' || train.ttnt === '1' ? 'Due / Arriving' : `${train.ttnt} min`; const isDirect = train.dest === 'NOP'; html += `
${dest}
${isDirect ? 'Direct to HK Island' : 'Shuttle to TIK / Transfer'} • Plat ${train.plat}
${ttnt}
`; }); html += '
'; contentDiv.innerHTML = html; } catch (e) { contentDiv.innerHTML = 'Error parsing live MTR data.'; } }, onerror: function() { contentDiv.innerHTML = 'Failed to connect to MTR Open Data API.'; } }); } const observer = new MutationObserver(() => { if (!document.getElementById('mtr-lohas-widget')) { createMTRWidget(); fetchSchedule(); } }); observer.observe(document.body, { childList: true, subtree: true }); setInterval(fetchSchedule, 30000); // 30s auto-refresh })();
